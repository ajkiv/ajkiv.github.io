    /**
      * X1 Heaviest tree.
      *
      * X1. Write an algorithm that finds the heaviest tree component of a
      * directed graph. The weight of a tree is the sum of the weights of its
      * edges. A legitimate tree is a separate full component of a graph that
      * has no back, forward, or cross edges (within the tree, or from
      * outside). Algorithm returns the weight of the heaviest tree, or
      * Float.NaN if there is not a single legitimate tree in the graph. The
      * heaviest tree is also placed to the list of vertices given as a
      * parameter, root as the first element of the list. What it the time
      * complexity of your algorithm?
      *
      * Do not start coding (or touch the keyboard) before you have a proper
      * plan and a working algorithm on paper. Make your solution modular,
      * solve the problem in stages.
      *
      * Hints: You need to first find the potential root vertices in the graph,
      * then check that the compo- nents of the roots really are distinct trees
      * and calculate the weights of the trees. Take a skeleton and test
      * program from course www-page. A partial solution is better than no
      * solution. If you make a partial solution, write the self evaluation
      * correspondingly.
      *
      * @param G the input directed graph
      * @param resultTree an empty list to which resulting heaviest tree will be stored if found.
      * @return Weight of the tree, or Float.NaN if no tree in graph.
      **/
    float heaviestTree(DiGraph G, LinkedList resultTree) {
    	
    	// Time complexity
    	// -n = number of nodes, e = number of edges
    	// -The nodes get colored with the base-color white numerous times, O(n)
    	// -rootVertices(DiGraph g):
    	//  *Visits all nodes, and loots at all neighbours of the nodes: O(n+e)
    	//  *Gathers all the nodes that get colored black by visiting all nodes: O(n)
    	// -Finding out all the authhentic trees:
    	//  *going through the list of rootnodes (twice): O(n) at worst case
    	//  *deleting non-authentic root nodes: worst case O(n), average case O(1)
    	// -Finding the heaviest tree:
    	//  *traversing through the trees using edges starting from root node
    	//   -worst case O(n+e)
        // -Building the heaviest tree into a LinkedList
    	//  *authentic trees have sparse edges
    	//  *authentic trees are likely to have less nodes than total nodes in the graph
    	//  *result: time complexity is less than in the heaviest part of the code, which was O(n+e)
    	// -Final time complexity: O(n+e)
    	//  *Heaviest part of the code is at rootVertices(DiGraph g): O(n+e)


    	// Variable for measuring weight of a tree
    	float treeWeight = 0;
    	
    	// Variable for keeping track of the heaviest tree found so far
        float heaviestTree = 0;
        
        LinkedList tempRoot = new LinkedList();

        LinkedList authenticRoots = new LinkedList();
        
        LinkedList inauthenticRoots = new LinkedList();
        
        // Initializing a Set for keeping track of root vertices
        Set initialRoots = new Set();

        // Using the rootVertices() method to find all the rootVertices in the graph
        initialRoots = rootVertices(G);

        // Adding the found root vertices to the juuret1 -list
        authenticRoots.addAll(initialRoots);
        
        // We traverse through the rootnodes-list in order for the first time
        // we call the notTree() method to know which roots are parts of inauthentic trees
        colorTheGraph(G, DiGraph.WHITE);
        for (int nro = 0; nro < authenticRoots.size(); nro++) {
        	if (invalidTree(authenticRoots.get(nro)) == true) {
        		inauthenticRoots.add(authenticRoots.get(nro));
        	}
        }

        // Traversing through the rootnodes-list in reverse order to find trees with 2 root nodes
        colorTheGraph(G, DiGraph.WHITE);
        for (int nro = 0; nro < authenticRoots.size(); nro++) {
        	if (invalidTree(authenticRoots.get((authenticRoots.size()-1)-nro))== true) {
        		inauthenticRoots.add(authenticRoots.get(nro));
        	}
        }

        // We remove from the rootnodes-list the rootnodes which don't contain an authentic tree
        for (int nro = 0; nro < authenticRoots.size(); nro++) {
            if (inauthenticRoots.contains(authenticRoots.get(nro))) {
            	authenticRoots.remove(nro);
            }
        }
        
        // If there are rootnodes left, we will start searching for the heaviest tree
        if (authenticRoots.peekFirst() != null) {
            for (int nro = 0; nro < authenticRoots.size(); nro++) {
            	colorTheGraph(G, DiGraph.WHITE);
                treeWeight = (measureTreeWeight(authenticRoots.get(nro)));
                if (treeWeight > heaviestTree) {
                	heaviestTree = treeWeight;
                	tempRoot.clear(); 
                	tempRoot.add(authenticRoots.get(nro));
                }
            }
            // Now we start building a LinkedList for the heaviest tree we found
            
            // We begin by coloring all the graph nodes white
            colorTheGraph(G, DiGraph.WHITE);
            
            // Case, where the only tree is the size of the rootnode
            if (tempRoot.peekFirst() == null) {
            	resultTree.add(authenticRoots.peekFirst());
            	return heaviestTree;
            }
            // Here we build the LinkedList of the heaviest tree
            else {
              addDfs(resultTree, tempRoot.get(0));
              return heaviestTree;
            }
        }

        else {
            return Float.NaN;
        }
    } // heaviestTree()

    // Dfs = depth first search
    void addDfs(LinkedList resultTree, Vertex start) {
        start.setColor(DiGraph.BLACK);
        resultTree.add(start);
        for (Vertex n : start.neighbors()) {
            if (n.getColor() == DiGraph.WHITE) {
                addDfs(resultTree, n);
            }
        }
    }

    float measureTreeWeight(Vertex start) {
    	float weightOfTree = 0;
    	//float puunPaino = 0;
    	start.setColor(DiGraph.BLACK);
        for (Edge e : start.edges()) {
            Vertex w = e.getEndPoint(start);
            if (w.getColor() == DiGraph.WHITE) {
            	weightOfTree = weightOfTree + e.getWeight();
            	weightOfTree = weightOfTree + measureTreeWeight(w);
            }
        }
        return weightOfTree;
        }

  boolean invalidTree(Vertex start) {  
	start.setColor(DiGraph.GREY);
	for (Vertex vertex : start.neighbors()) {
		if (vertex.getColor() == DiGraph.GREY){
			return true;
		}
		else if (vertex.getColor() == DiGraph.BLACK) {
			return true;
		}
		else if (vertex.getColor() == DiGraph.WHITE) {
			if(invalidTree(vertex)) {
				return true;
			}
		}
	}
	start.setColor(Graph.BLACK);
	return false;
}

    // Set of vertices with no incoming edge
    Set rootVertices(DiGraph g) {

        Set R = new Set();

        colorTheGraph(g, DiGraph.WHITE);
        
        // We color the non-rootnodes black
        for (Vertex v : g.vertices()) {
            if (v.getColor() == 1);
            rootColorDfs(v);
        }

        // Collecting rootnodes (nodes colored white)
        for (Vertex v : g.vertices()) {
            if (v.getColor() == 1) {
                R.add(v);
            }
        }

        return R;
    }   // rootVertices()


    void rootColorDfs(Vertex start) {
        for (Vertex n : start.neighbors()) {
            if (n.getColor() == 1) {
                n.setColor(2);
                rootColorDfs(n);
            }
        }
    }

    void colorTheGraph(DiGraph G, int color) {
        for (Vertex v : G.vertices()) {
            v.setColor(color);
        }
    }
}   // class
