    void addDfs(LinkedList resultTree, Vertex start) {
        start.setColor(DiGraph.BLACK);
        resultTree.add(start);
        for (Vertex n : start.neighbors()) {
            if (n.getColor() == DiGraph.WHITE) {
                addDfs(resultTree, n);
            }
        }
    }

    float puunPaino(Vertex start) {
    	float puunPaino = 0;
    	start.setColor(DiGraph.BLACK);
        for (Edge e : start.edges()) {
            Vertex w = e.getEndPoint(start);
            if (w.getColor() == DiGraph.WHITE) {
                puunPaino = puunPaino + e.getWeight();
                puunPaino = puunPaino + puunPaino(w);
            }
        }
        return puunPaino;
        }

  boolean eiPuu(Vertex start) {
	start.setColor(DiGraph.GREY);
	for (Vertex vertex : start.neighbors()) {
		if (vertex.getColor() == DiGraph.GREY){
			return true;
		}
		else if (vertex.getColor() == DiGraph.BLACK) {
			return true;
		}
		else if (vertex.getColor() == DiGraph.WHITE) {
			if(eiPuu(vertex)) {
				return true;
			}
		}
	}
	start.setColor(Graph.BLACK);
	return false;
}

    Set rootVertices(DiGraph g) {

        }

            }

    }   // rootVertices()


    void rootColorDfs(Vertex start) {
        for (Vertex n : start.neighbors()) {
            if (n.getColor() == 1) {
                n.setColor(2);
                rootColorDfs(n);
            }
        }
    }

    void varita(DiGraph G, int vari) {
        for (Vertex v : G.vertices()) {
            v.setColor(vari);
        }
