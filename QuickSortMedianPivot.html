public class QuickSortMedianPivot {
    public static void main(String[] args) {  
        int[] array = {3, 6, 8, 10, 1, 1, 10, 2, 2, 2, 2, 2, 1, 8, 6, 10};
        quicksort(array, 0, array.length - 1);
        
        // Print the sorted array
        for (int i : array) {
            System.out.print(i + " ");
        }
        System.out.println();
    }
    
    public static void quicksort(int[] array, int left, int right) {
      // check for base case in recursion
      if(right <= left) {
        return;
      }
      
      int pivotIndex = choosePivot(array, left, right);
      int pivotValue = array[pivotIndex];
      
      // Move pivot to the end
      swap(array, pivotIndex, right); // This modifies the original array directly
      pivotIndex = right; // Now the pivot is at the end
      
      // Partition the array around the pivotValue
      int newPivotIndex = partition(array, left, right, pivotValue);

      // Recursive calls for left and right sub-arrays
      quicksort(array, left, newPivotIndex - 1);
      quicksort(array, newPivotIndex + 1, right);
    }
    
    // Method to choose pivot using the median-of-three method
    private static int choosePivot(int[] array, int left, int right) {
        int mid = left + (right - left) / 2;

        // Use the first, middle, and last values for comparison
        int first = array[left];
        int middle = array[mid];
        int last = array[right];

        // Find the median of the three
        if ((first >= middle && first <= last) || (first <= middle && first >= last)) {
            return left; // first is the median
        } else if ((middle >= first && middle <= last) || (middle <= first && middle >= last)) {
            return mid; // middle is the median
        } else {
            return right; // last is the median
        }
    }
    
    // swap method directly influences the array
    private static void swap(int[] array, int i, int j) {
        int temp = array[i];
        array[i] = array[j];
        array[j] = temp;
    }
    
    private static int partition(int[] array, int left, int right, int partitionPivotValue) {
        int storeIndex = left;
        for (int i = left; i < right; i++) {
            if (array[i] < partitionPivotValue) {
                swap(array, storeIndex, i);
                storeIndex++;
            }
        }
        // Move pivot to its final place
        swap(array, storeIndex, right);
        return storeIndex;
    }
}  

// runtime complexity:
// best case: O(n log n)
// average case: O(n log n) 
// worst case (array is almost sorted): O(n^2)

// note: using median-of-three -pivot leads to more balanced splits and 
//      reduced depth of recursion,
//      but the Ordo-complexities in all 3 cases (best, average, worst) remain the same.

// space complexity: O()log(n)) because of recursion
