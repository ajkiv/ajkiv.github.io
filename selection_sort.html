import random

def main():
    # Parameters for creating an array of unsorted numbers
    size = 10  # Number of random numbers
    lower_bound = 1  # Minimum value
    upper_bound = 100  # Maximum value
    
    # Generate the random list
    unsorted_array = generate_random_numbers(size, lower_bound, upper_bound)

    # Print the generated random numbers
    print("Unsorted array of random numbers: ", unsorted_array)

    sorted_array = selection_sort(unsorted_array)
    
    # Print the array that was sorted
    print("Sorted Array:", sorted_array)

def selection_sort(arr):    
    # Get the length of the array
    n = len(arr)
    
    # Traverse through all array elements
    for i in range(n):
        # Assume the first element of the unsorted section is the minimum
        min_index = i
        
        # Check the elements in the unsorted section
        for j in range(i + 1, n):
            # Update the index of the minimum element if a smaller element is found
            if arr[j] < arr[min_index]:
                min_index = j
        
        # Swap the smallest found element with the first element of the unsorted section
        if min_index != i:  # Only swap if a new minimum was found
            swap1(arr, i, min_index)

    return arr

def swap1(arr, i, j):
    # Swaps the elements at index i and j in the array.
    arr[i], arr[j] = arr[j], arr[i]

def swap2(arr, i, j):
    # Swaps the elements at index i and j in the array.
    temp = arr[i]
    arr[i] = arr[j]
    arr[j] = temp

# Function to generate a list of random unsorted numbers
def generate_random_numbers(size, lower_bound, upper_bound):
    return [random.randint(lower_bound, upper_bound) for _ in range(size)]

if __name__ == "__main__":
    main()

# Example run:
#
# Unsorted array of random numbers:  [91, 49, 21, 87, 12, 21, 54, 43, 6, 50]
# Sorted Array: [6, 12, 21, 21, 43, 49, 50, 54, 87, 91]

# Time complexity:
#
# Best case: O(n) (checks the list through even if the list is already sorted)
# Average case: O(n)
# Worst case: O(n) (List in reverse order, but number of comparisons remains the same)
